//===== TERMUXFILE: start_jarvis_services.sh =====
#!/data/data/com.termux/files/usr/bin/bash
WORKFLOWS=~/workflows
AGENT_DIR=$WORKFLOWS/test/jarvis-agent
LOG_DIR=$WORKFLOWS

echo "[services] Stopping old instances..."
pkill -f agent_server.py        2>/dev/null
pkill -f elevenlabs_tts.py      2>/dev/null
pkill -f jarvis_conversation.py 2>/dev/null
sleep 1

# Load Gemini key — exported so child processes inherit it
GEMINI_KEY_FILE=$WORKFLOWS/.gemini_key
if [ -f "$GEMINI_KEY_FILE" ]; then
    export GEMINI_API_KEY=$(cat "$GEMINI_KEY_FILE" | tr -d '[:space:]')
    echo "[services] Gemini key loaded"
else
    echo "[services] ⚠  No .gemini_key file found at $GEMINI_KEY_FILE"
fi

echo "[services] Starting agent_server (8891)..."
nohup python3 $AGENT_DIR/agent_server.py \
    > $LOG_DIR/agent_server.log 2>&1 &
sleep 1

echo "[services] Starting elevenlabs_tts (8892)..."
nohup python3 $WORKFLOWS/elevenlabs_tts.py \
    > $LOG_DIR/tts_server.log 2>&1 &
sleep 1

echo "[services] Starting jarvis_conversation (8893)..."
nohup python3 $AGENT_DIR/jarvis_conversation.py \
    > $LOG_DIR/conversation_server.log 2>&1 &
sleep 2

echo "[services] Health checks..."
curl -sf http://127.0.0.1:8891/health && echo " agent_server      OK" || echo " agent_server      FAILED"
curl -sf http://127.0.0.1:8892/health && echo " elevenlabs_tts    OK" || echo " elevenlabs_tts    FAILED"
curl -sf http://127.0.0.1:8893/health && echo " jarvis_conv       OK" || echo " jarvis_conv       FAILED"
echo "[services] Done."
