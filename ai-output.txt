package com.jarvismini.core.progress

import android.content.Context

object ProgressStore {

    private val completedBlocks = mutableSetOf<String>()

    fun markComplete(context: Context, blockId: String) {
        completedBlocks.add(blockId)
    }

    fun markIncomplete(context: Context, blockId: String) {
        completedBlocks.remove(blockId)
    }

    fun getTodayBlocks(context: Context): List<ProgressBlock> {
        return completedBlocks.map { id ->
            ProgressBlock(
                id = id,
                completed = true
            )
        }
 
}

}

package com.jarvismini.core.progress

import android.content.Context
import java.text.SimpleDateFormat
import java.util.*

object ProgressStatsEngine {

    fun getTodayStats(context: Context): ProgressStats {
        val blocks = ProgressRepository.getTodayBlocks(context)
        val total = blocks.size
        val completedCount = blocks.count { block -> block.completed }

        val today = SimpleDateFormat(
            "yyyy-MM-dd",
            Locale.getDefault()
        ).format(Date())

        return ProgressStats(
            date = today,
            totalBlocks = total,
            completedBlocks = completedCount
        )
  
}
}
