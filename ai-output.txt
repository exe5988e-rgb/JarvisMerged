package com.jarvismini.automation

import android.accessibilityservice.AccessibilityService
import android.view.accessibility.AccessibilityNodeInfo
import java.util.ArrayDeque

object AccessibilityServiceHelper {

    // The active AccessibilityService instance
    lateinit var service: AccessibilityService

    /**
     * Initialize with the running AccessibilityService
     */
    fun init(s: AccessibilityService) {
        service = s
    }

    /**
     * Returns the root node of the active window
     * Useful for notification/chat automation
     */
    fun getRootNode(): AccessibilityNodeInfo? {
        return service.rootInActiveWindow
    }

    /**
     * Handles OnePlus Clock automation for WorkMode:
     * - Navigates to Stopwatch tab
     * - Clicks Start if not running
     * - Clicks Pause/Stop if already running
     */
    fun handleOnePlusClock() {
        val root = service.rootInActiveWindow ?: return

        // Navigate to Stopwatch tab
        clickByText(root, "Stopwatch")

        // Click Pause or Stop if running
        if (clickByText(root, "Pause") || clickByText(root, "Stop")) return

        // Otherwise click Start
        clickByText(root, "Start")
    }

    /**
     * Recursive search for clickable node by visible text
     * @param root Root node to search
     * @param target Text to match
     * @return true if click performed
     */
    private fun clickByText(root: AccessibilityNodeInfo, target: String): Boolean {
        val stack = ArrayDeque<AccessibilityNodeInfo>()
        stack.add(root)

        while (stack.isNotEmpty()) {
            val node = stack.removeFirst()
            val text = node.text?.toString()?.lowercase()

            if (node.isClickable && text == target.lowercase()) {
                node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
                return true
            }

            // Add children to stack
            for (i in 0 until node.childCount) {
                node.getChild(i)?.let { stack.add(it) }
            }
        }
        return false
    }
}
