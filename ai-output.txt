#!/usr/bin/env python3
"""
set_ports.py — JarvisMinI Port Configuration Tool
Run on either Phone A or Phone B — patches all relevant files.

Usage:
  python3 ~/set_ports.py

Ports explained:
  Phone A ADB port  — changes on reboot (e.g. 39427)
  Phone B ADB port  — changes on reboot (e.g. 36429) → goes into claude_bridge.py SERIAL
  Phone B HTTP port — FIXED at 8890, never changes    → goes into llm_client.py PHONE_B_URL

Session 10 fix:
  - Separated Phone B ADB port (SERIAL in bridge) from Phone B HTTP port (URL in llm_client).
    Old script conflated them — setting bridge port to 8890 instead of ADB serial.
  - Added workflows/claude_bridge.py to PATCHES.
  - Fixed no-match issues by checking actual file content for variable names.

Session 20 fix:
  - Added self_repair.py PHONE_B_ADB patch so start_agent_task() ADB push
    uses the correct port after every reboot.
"""

import os
import re
import sys

HOME = os.path.expanduser("~")

R   = "\033[1;31m"
G   = "\033[1;32m"
Y   = "\033[1;33m"
B   = "\033[1;34m"
C   = "\033[1;36m"
W   = "\033[1;37m"
DIM = "\033[2m"
RST = "\033[0m"
LINE = f"{DIM}{'─'*54}{RST}"

BRIDGE_HTTP_PORT = "8890"  # fixed, never changes

def banner():
    os.system("clear")
    print(f"""
{C}╔══════════════════════════════════════════════════════╗
║     JARVIS MINI — PORT CONFIGURATION TOOL           ║
║     Updates all config files in one shot            ║
╚══════════════════════════════════════════════════════╝{RST}
""")

def read_current():
    """Read current ports from existing files."""
    a_ip, a_port   = "192.168.29.48",  "39427"
    b_ip, b_adb    = "192.168.29.110", "36429"

    # Phone B ADB serial from claude_bridge.py
    for bridge_path in ["claude_bridge.py", "workflows/claude_bridge.py"]:
        full = os.path.join(HOME, bridge_path)
        if os.path.exists(full):
            m = re.search(r'SERIAL\s*=\s*"([^":]+):(\d+)"', open(full).read())
            if m:
                b_ip   = m.group(1)
                b_adb  = m.group(2)
            break

    # Phone A serial from llm_client.py
    for llm_path in [
        "test/jarvis-agent/reasoning/llm_client.py",
        "workflows/test/jarvis-agent/reasoning/llm_client.py",
    ]:
        full = os.path.join(HOME, llm_path)
        if os.path.exists(full):
            m = re.search(r'PHONE_A_SERIAL\s*=\s*"([^":]+):(\d+)"', open(full).read())
            if m:
                a_ip  = m.group(1)
                a_port = m.group(2)
            break

    return a_ip, a_port, b_ip, b_adb

def ask(prompt, default):
    val = input(f"  {prompt} [{default}]: ").strip()
    return val if val else default

def patch_file(path, replacements):
    """Apply list of (pattern, replacement) to file. replacement uses {A_IP} etc."""
    full = os.path.join(HOME, path)
    if not os.path.exists(full):
        print(f"  {DIM}~  not found — {path}{RST}")
        return
    text = open(full).read()
    original = text
    for label, pattern, repl in replacements:
        new = re.sub(pattern, repl, text)
        if new != text:
            print(f"  {G}✓{RST}  {label:<26} {DIM}{path}{RST}")
            text = new
        else:
            print(f"  {DIM}~  {label:<26} no match — {path}{RST}")
    if text != original:
        open(full, "w").write(text)

def patch_sh(path, a_ip, a_port, b_ip, b_adb):
    """Replace all IP:port pairs in shell scripts."""
    full = os.path.join(HOME, path)
    if not os.path.exists(full):
        return
    text = open(full).read()
    original = text
    pairs = list(dict.fromkeys(re.findall(r'(192\.168\.\d+\.\d+):(\d+)', text)))
    if not pairs:
        print(f"  {DIM}~  no IP:port found — {path}{RST}")
        return
    if len(pairs) >= 2:
        text = text.replace(f"{pairs[0][0]}:{pairs[0][1]}", f"{a_ip}:{a_port}")
        text = text.replace(f"{pairs[1][0]}:{pairs[1][1]}", f"{b_ip}:{b_adb}")
    elif len(pairs) == 1:
        text = text.replace(f"{pairs[0][0]}:{pairs[0][1]}", f"{a_ip}:{a_port}")
    if text != original:
        open(full, "w").write(text)
        print(f"  {G}✓{RST}  ports updated          {DIM}{path}{RST}")
    else:
        print(f"  {DIM}~  no changes               {path}{RST}")

def main():
    banner()

    a_ip, a_port, b_ip, b_adb = read_current()

    print(f"  {W}Current:{RST}")
    print(f"  {B}Phone A{RST}  ADB: {a_ip}:{a_port}")
    print(f"  {C}Phone B{RST}  ADB: {b_ip}:{b_adb}  HTTP: {b_ip}:{BRIDGE_HTTP_PORT} (fixed)")
    print(f"\n{LINE}\n")
    print(f"  Press {G}Enter{RST} to keep current value\n")

    print(f"{B}  PHONE A — OPPO (agent loop){RST}\n")
    a_ip   = ask("IP  ", a_ip)
    a_port = ask("ADB Port", a_port)
    print()

    print(LINE)
    print(f"\n{C}  PHONE B — Samsung Galaxy A10s (bridge){RST}\n")
    b_ip   = ask("IP  ", b_ip)
    b_adb  = ask("ADB Port (changes on reboot)", b_adb)
    print()

    print(LINE)
    print(f"\n  {W}Summary:{RST}\n")
    print(f"  {B}Phone A{RST}  {a_ip}:{a_port}")
    print(f"  {C}Phone B{RST}  ADB={b_ip}:{b_adb}  HTTP={b_ip}:{BRIDGE_HTTP_PORT}")
    print()
    ans = input(f"  Apply? [{G}Y{RST}/{R}n{RST}]: ").strip().lower()
    if ans not in ("", "y", "yes"):
        print(f"\n  {Y}Cancelled.{RST}\n")
        sys.exit(0)

    print(f"\n{LINE}")
    print(f"  {W}Applying...{RST}\n")

    # ── claude_bridge.py: SERIAL = Phone B ADB ──────────────────────────────
    for path in ["claude_bridge.py", "workflows/claude_bridge.py"]:
        patch_file(path, [
            ("SERIAL (Phone B ADB)",
             r'(SERIAL\s*=\s*")[^"]+(")',
             f'\\g<1>{b_ip}:{b_adb}\\g<2>'),
        ])

    # ── controller.py: DEFAULT_DEVICE = Phone A ADB ─────────────────────────
    for path in [
        "test/jarvis-agent/agent/controller.py",
        "workflows/test/jarvis-agent/agent/controller.py",
    ]:
        patch_file(path, [
            ("DEFAULT_DEVICE (Phone A)",
             r'(DEFAULT_DEVICE\s*=\s*")[^"]+(")',
             f'\\g<1>{a_ip}:{a_port}\\g<2>'),
        ])

    # ── llm_client.py: PHONE_A_SERIAL + PHONE_B_URL (HTTP port stays 8890) ──
    for path in [
        "test/jarvis-agent/reasoning/llm_client.py",
        "workflows/test/jarvis-agent/reasoning/llm_client.py",
    ]:
        patch_file(path, [
            ("PHONE_A_SERIAL",
             r'(PHONE_A_SERIAL\s*=\s*")[^"]+(")',
             f'\\g<1>{a_ip}:{a_port}\\g<2>'),
            ("PHONE_B_URL",
             r'(PHONE_B_URL\s*=\s*"http://)[^"]+(")',
             f'\\g<1>{b_ip}:{BRIDGE_HTTP_PORT}\\g<2>'),
        ])

    # ── self_repair.py: PHONE_B_ADB — used by start_agent_task() ADB push ───
    # Session 20: self_repair.py has its own PHONE_B_ADB constant that must
    # match the bridge serial. Without this patch, the ADB push step of
    # start_agent_task() sends new_dump.txt to the wrong port after reboot.
    for path in [
        "test/jarvis-agent/agent/self_repair.py",
        "workflows/test/jarvis-agent/agent/self_repair.py",
    ]:
        patch_file(path, [
            ("PHONE_B_ADB (self_repair)",
             r'(PHONE_B_ADB\s*=\s*")[^"]+(")',
             f'\\g<1>{b_ip}:{b_adb}\\g<2>'),
        ])

    # ── shell scripts ────────────────────────────────────────────────────────
    for path in ["jarvis_server_init.sh", "workflows/jarvis_server_init.sh",
                 "workflows/jarvis_start.sh"]:
        patch_sh(path, a_ip, a_port, b_ip, b_adb)

    print(f"\n{LINE}")
    print(f"\n  {G}Done.{RST}\n")
    print(f"  {DIM}Restart bridge on Phone B:{RST}")
    print(f"  adb -s {b_ip}:{b_adb} shell am startservice ca.zgrs.clipper/.ClipboardService")
    print(f"  pkill -f claude_bridge.py && python3 ~/claude_bridge.py\n")

if __name__ == "__main__":
    main()
