"""
agent/controller.py  —  Jarvis Agent Controller

Session 15 (OPT 12): Hierarchical Task Planner wired in.
  - TaskPlanner created in run() after health check
  - planner.tick() called at start of every run_step()
  - planner.current_goal_description() passed to build_prompt() as sub_goal=
  - policy engine also receives sub-goal task string for better matching
  - state.sub_goal_label / sub_goal_index updated each step for logging
  - planner.done() checked alongside state.is_finished()

Session 15 (OPT 13): Bridge auto-recovery wired into llm_client.py

Session 15 (multi-AI summarize):
  - save_clipboard goal: runs shell research_<app> = termux-clipboard-get
    immediately after copy_ai_response — no LLM call, no step waste
  - summarize_memory goal: collects all research_* keys from memory,
    calls Groq with a merge prompt, stores result in memory["summary"]
  - paste_and_send uses {summary} from memory when planner.has_summarize()

Session 15 fix (OPT 13):
  - wait action handled in run_step(): sleep 4s, re-screenshot, no dispatch
    Prevents retry loops when Phone B returns 'wait' from bridge timeout
    or when action_parser converts 'Sending...' → wait via ui_echo filter

Session 16 fix (stale UI + wait loop):
  - OPT7 prediction hit: validate predicted screen activity matches live
    activity before serving cached elements. If mismatch → fresh dump.
    Prevents popup/stale elements from a previous Gemini visit being served
    as the current Gemini screen.
  - wait action now calls invalidate_cache(device) before sleeping so the
    next step always gets a fresh UI dump instead of the same stale screen.
  - type action now also calls invalidate_cache(device) after execution.
    Typing changes the screen (send button appears, suggestions load) but
    the activity string stays the same — OPT7 was serving the stale
    pre-type cache, hiding the Send button entirely.
  - invalidate_cache now also called in the error retry handler. When
    Phone B drops mid-step the tap fires but invalidate never ran —
    next step would serve stale cache. Now cleared on every exception.

Session 17: Pause gate added.
  - run_step() checks is_paused() at the very top, blocking until resumed.
  - PAUSE_FLAG = ~/workflows/.jarvis_pause (same path used by agent_server).
"""

def is_paused() -> bool:

def get_top_activity(device: str) -> str:
    import subprocess
    try:
        r = subprocess.run(
            ["adb", "-s", device, "shell", "dumpsys", "activity", "activities"],
            capture_output=True, text=True, timeout=8
        )
        import re
        m = re.search(
            r'\*\s*Hist\s*#\d+:\s*ActivityRecord\{[^ ]+\s*[^ ]+\s*([^ ]+)\s*t\d+\}',
            r.stdout
        )
        if m:
            return m.group(1)
    except Exception as e:
        print(f"[controller] activity error: {e}")
    return ""

def extract_apps_in_order(task: str) -> list:
    import re
    task_lc = task.lower()
    found   = []
    for app, pkg in APP_PACKAGES.items():
        if app in task_lc and app not in found:
            found.append(app)
    return found

def _screen_name(activity: str) -> str:
    if not activity:
        return "unknown"
    parts = activity.split("/")
    return parts[-1].replace("Activity", "").lower() if parts else activity

def _action_label(action: dict, ui: list) -> str:
    act = action.get("action", "?")
    if act == "tap":
        tid = action.get("target")
        if isinstance(tid, int):
            label = next((e["text"][:30] for e in ui if e["id"] == tid), f"id={tid}")
            return f"tap:{label}"
        return f"tap:({action.get('x')},{action.get('y')})"
    if act == "type":
        return f"type:{action.get('text','')[:20]}"
    return act

def _is_on_launcher(device: str) -> bool:
    act = get_top_activity(device)
    return "launcher" in act.lower() or "home" in act.lower()

def _is_on_ai_screen(ui: list) -> bool:
    return any(
        any(frag in el.get("package", "").lower() for frag in _AI_PKG_FRAGMENTS)
        for el in ui
    )

def _compile_research(memory: dict, task: str) -> str:
    lines = ["Research Report", f"Task: {task}", "=" * 40, ""]
    for key, value in memory.items():
        label = key.replace("_", " ").title()
        lines.append(f"{label}:")
        lines.append(str(value))
        lines.append("")
    return "\n".join(lines)

def _summarize_memory(memory: dict, task: str, model_path: str) -> str:
    """
    OPT 13 — Collect all research_* keys from memory, call LLM to merge
    into a single concise summary. Stores result in memory["summary"].
    Returns the summary string.
    """
    research_keys = {k: v for k, v in memory.items() if k.startswith("research_")}
    if not research_keys:
        print("[summarize] no research_* keys found in memory — skipping")
        return ""

    sections = []
    for key, value in research_keys.items():
        app_name = key.replace("research_", "").capitalize()
        sections.append(f"=== {app_name} ===\n{value}")

    combined = "\n\n".join(sections)
    prompt = (
        f"You are a research assistant. The following are research results from multiple AI apps "
        f"on the topic: \"{task}\"\n\n"
        f"{combined}\n\n"
        f"Merge these into ONE concise summary (max 300 words). "
        f"Output ONLY the summary text, no headers, no preamble."
    )

    print(f"[summarize] calling LLM to merge {len(research_keys)} research results...")
    try:
        summary = generate(prompt, model_path=model_path, max_tokens=400)
        summary = summary.strip()
        print(f"[summarize] summary ({len(summary)} chars): {summary[:100]!r}")
        memory["summary"] = summary
        return summary
    except Exception as e:
        print(f"[summarize] LLM call failed: {e}")
        fallback = combined[:1000]
        memory["summary"] = fallback
        return fallback

def _clipboard_via_server() -> str:
    import urllib.request, json
    try:
        import subprocess
        r = subprocess.run(
            ["termux-clipboard-get"],
            capture_output=True, text=True, timeout=5
        )
        text = r.stdout.strip()
        if text:
            print(f"[clipboard] subprocess success: {text[:60]!r}")
            return text
    except Exception as e:
        print(f"[clipboard] subprocess failed: {e}")
    try:
        url = f"http://127.0.0.1:{AGENT_SERVER_PORT}/clipboard"
        with urllib.request.urlopen(url, timeout=6) as resp:
            data = json.loads(resp.read().decode())
            text = data.get("text", "")
            if text:
                print(f"[clipboard] /clipboard success: {text[:60]!r}")
            return text
    except Exception as e:
        print(f"[clipboard] /clipboard failed: {e}")
        return ""

def _resolve_action(action: dict, ui: list, memory: dict,
                    task: str = "", device: str = "") -> dict:
    act = action.get("action")

    if act == "extract":
        key        = action.get("key", "")
        value_hint = action.get("value", "")
        if key in memory and len(str(memory[key])) > 60:
            print(f"[memory] extract {key!r} BLOCKED — shell already set {len(str(memory[key]))} chars")
            return {**action, "resolved_value": memory[key]}
        is_ai_screen = any(
            any(frag in el.get("package", "").lower() for frag in _AI_PKG_FRAGMENTS)
            for el in ui
        )
        if is_ai_screen and key in memory and str(memory.get(key, "")).strip() == "":
            copy_el = next((e for e in ui if e.get("text", "").strip() == "Copy"), None)
            if copy_el:
                print(f"[memory] extract {key!r} on AI screen — redirecting to tap Copy")
                return {"action": "tap", "target": copy_el["id"]}
        resolved = extract_from_ui(key, ui, value_hint)
        print(f"[memory] extract {key!r} = {resolved!r}")
        return {**action, "resolved_value": resolved or ""}

    if act == "type":
        t           = action.get("text", "")
        substituted = substitute_memory(t, memory)
        if substituted != t:
            print(f"[memory] substituted {t!r} → {substituted!r}")
        return {**action, "text": substituted}

    if act == "shell":
        import subprocess
        key     = action.get("key", "result")
        command = action.get("command", "")
        if "termux-clipboard-get" in command:
            print(f"[shell] clipboard → routing via _clipboard_via_server()")
            output = _clipboard_via_server()
            print(f"[shell] clipboard result: {output[:100]!r}")
            if not output:
                print(f"[shell] WARNING: clipboard empty — Copy tap may not have registered")
            return {**action, "resolved_value": output, "key": key}
        result = subprocess.run(
            command, shell=True,
            capture_output=True, text=True, timeout=30
        )
        output = result.stdout.strip()
        if not output and result.stderr:
            output = result.stderr.strip()
        print(f"[shell] {command!r} → {output[:100]!r}")
        return {**action, "resolved_value": output, "key": key}

    if act == "share_file":
        import subprocess
        filename = action.get("filename", "research.txt")
        filepath = f"/sdcard/{filename}"
        cmd = (
            f"am start -a android.intent.action.SEND "
            f"-t text/plain "
            f"--eu android.intent.extra.STREAM file://{filepath} "
            f"--ez android.intent.extra.REPLACE_EXTRAS false "
            f"com.whatsapp"
        )
        result = subprocess.run(cmd, shell=True, capture_output=True,
                                text=True, timeout=10)
        print(f"[share] intent fired: {result.stdout.strip() or 'ok'}")
        return {**action, "resolved_path": filepath}

    if act == "save_file":
        import subprocess, base64 as _b64
        filename = action.get("filename", "research.txt")
        content  = action.get("content") or _compile_research(memory, task)
        filepath = f"/sdcard/{filename}"
        dev      = device or DEFAULT_DEVICE
        encoded  = _b64.b64encode(content.encode("utf-8")).decode("ascii")
        subprocess.run(["adb", "-s", dev, "shell",
                        f"echo '{encoded}' | base64 -d > {filepath}"],
                       capture_output=True, timeout=15)
        print(f"[file] saved {len(content)} chars → {filepath}")
        return {**action, "resolved_path": filepath}

    return action


# ── Step ──────────────────────────────────────────────────────────────────────

def run_step(state: AgentState, device: str, model_path: str,
             max_skill_steps: int = 0, planner: TaskPlanner = None):

    # ── Session 17: pause gate ────────────────────────────────────────────────
    while is_paused():
        print("[controller] ⏸ PAUSED — waiting for resume...")
        time.sleep(2)
    # ─────────────────────────────────────────────────────────────────────────

    print(f"\n{'='*56}\n  STEP {state.step+1}  |  {state.task}\n{'='*56}")

    img      = capture_screen(adb_device=device)
    state.last_screen = img

    activity   = get_top_activity(device)
    ui         = None
    current_id = None

    if state.last_screen_id and state.last_action_label:
        predicted_id = _tm.get_predicted_screen(
            state.last_screen_id, state.last_action_label
        )
        if predicted_id:
            predicted_screen = _sm.get_screen(predicted_id)
            if predicted_screen and predicted_screen.get("elements"):
                # Session 16 fix: validate predicted screen activity matches
                # current live activity. A stale popup screen stored under
                # the same Gemini activity would otherwise get served here.
                predicted_activity = predicted_screen.get("activity", "")
                activity_match = (not predicted_activity or
                                  not activity or
                                  predicted_activity == activity)
                if not activity_match:
                    print(f"[OPT7] activity mismatch: predicted={predicted_activity!r} "
                          f"live={activity!r} — skipping cache, fresh dump")
                elif any(frag in activity for frag in _AI_PKG_FRAGMENTS):
                    # AI app screens change every step (input state, response
                    # loading, send button) — never serve from cache
                    print(f"[OPT7] AI app screen — always fresh dump")
                elif predicted_screen.get("visits", 0) >= 10:
                    cached_els = predicted_screen["elements"]
                    print(f"[OPT7] prediction hit: {predicted_screen['name']} "
                          f"({predicted_screen.get('visits',0)} visits) — skipping dump")
                    ui = [dict(e, id=i+1) if "id" not in e else e
                          for i, e in enumerate(cached_els)]
                    texts = [e.get("text", "") for e in ui]
                    if any(len(t) > 200 or "\n[controller]" in t
                           or "\n[ui_parser]" in t for t in texts):
                        print("[OPT7] cached UI looks stale/corrupt — doing fresh dump")
                        ui = None
                    else:
                        current_id = predicted_id

    if ui is None:
        ui = parse_ui(img, adb_device=device)
        current_id = _sm.identify(activity, ui) if activity else None

    state.last_ui = ui
    print(f"[step] UI: {[e['text'] for e in ui[:8]]}")

    if current_id and state.last_screen_id and state.last_action_label:
        _tm.record(state.last_screen_id, state.last_action_label, current_id)
        _sm.record_transition(state.last_screen_id, state.last_action_label, current_id)

    _tm.clear_pending()

    if current_id:
        screen_info = _sm.get_screen(current_id)
        if screen_info:
            print(f"[map] screen: {screen_info['name']} ({current_id[:8]})")

    # OPT 12: Tick planner
    if planner and not planner.is_single_goal():
        planner.tick(activity, ui, state.history, state.memory)
        goal = planner.current_goal()
        if goal:
            state.sub_goal_label = goal.type
            state.sub_goal_index = planner.plan.index
            planner.print_status()

            # ── save_clipboard goal: execute inline, no LLM call ─────────────
            if goal.type == "save_clipboard":
                print(f"[PLANNER] executing save_clipboard → key={goal.key}")
                output = _clipboard_via_server()
                if output:
                    state.memory[goal.key] = output
                    print(f"[PLANNER] save_clipboard: stored {len(output)} chars as {goal.key!r}")
                    state.record({"action": "shell", "key": goal.key,
                                  "command": "termux-clipboard-get"},
                                 ui)
                else:
                    print(f"[PLANNER] save_clipboard: clipboard empty — will retry next step")
                state.step += 1
                return

            # ── summarize_memory goal: call LLM inline ───────────────────────
            if goal.type == "summarize_memory":
                print(f"[PLANNER] executing summarize_memory")
                summary = _summarize_memory(state.memory, state.task, model_path)
                if summary:
                    state.record({"action": "shell", "key": "summary",
                                  "command": "summarize_memory"}, ui)
                state.step += 1
                return

        if planner.done():
            print("[PLANNER] all sub-goals complete → task done")
            state.done = True
            return

    # Determine effective task for policy/skill/LLM
    if planner and not planner.is_single_goal() and not planner.done():
        effective_task = planner.current_goal_description()
        sub_goal_desc  = effective_task
    else:
        effective_task = state.task
        sub_goal_desc  = None

    # If summarize_memory was in plan, substitute {summary} in paste_and_send
    if planner and planner.has_summarize() and "summary" in state.memory:
        state.memory.setdefault("_paste_override", state.memory["summary"])

    # OPT 4 v2
    if state.app_index < len(state.task_apps):
        next_app = state.task_apps[state.app_index]
        pkg      = APP_PACKAGES.get(next_app, "")
        is_ai_app = any(frag in pkg for frag in _AI_PKG_FRAGMENTS)
        if not is_ai_app and state.app_index > 0:
            prev_app     = state.task_apps[state.app_index - 1]
            prev_pkg     = APP_PACKAGES.get(prev_app, "")
            prev_was_ai  = any(frag in prev_pkg for frag in _AI_PKG_FRAGMENTS)
            if prev_was_ai:
                already_copied = any("Copy" in h and "tap" in h for h in state.history)
                if not already_copied:
                    print(f"[OPT4v2] blocked — {prev_app} Copy not done yet, falling through to policy")

        if _is_on_launcher(device) and not _is_on_ai_screen(ui):
            print(f"[OPT4v2] launcher -> launching {next_app} ({pkg})")
            generate("", model_path=model_path, bypass_pkg=pkg)
            label = f"opt4_launch:{next_app}"
            state.history.append(label)
            state.last_action_label = label
            state.last_screen_id    = current_id
            state.step += 1
            state.app_index += 1
            invalidate_cache(device)
            _tm.set_pending(current_id, label)
            return
        elif _is_on_ai_screen(ui):
            print(f"[OPT4v2] blocked — still on AI app screen, waiting for capture")

    # Empty UI
    if not ui:
        state.empty_ui_count += 1
        print(f"[step] empty UI ({state.empty_ui_count}/{MAX_EMPTY_WAITS})")
        if state.empty_ui_count >= MAX_EMPTY_WAITS:
            state.done = True
            return
        time.sleep(2)
        state.step += 1
        return
    state.empty_ui_count = 0

    # Policy Engine
    policy_actions = get_policy_actions(
        effective_task, activity, ui, state.history, state.memory
    )
    if policy_actions:
        print(f"[policy] matched — executing {len(policy_actions)} actions")
        policy_done = False
        for action in policy_actions:
            action = _resolve_action(action, ui, state.memory,
                                     task=state.task, device=device)
            print(f"[policy] action: {action}")
            execute(action, ui, adb_device=device)
            label = _action_label(action, ui)
            state.record(action, ui)
            state.last_action_label = label
            state.last_screen_id    = current_id
            if action.get("action") == "done":
                policy_done = True
                break
        _tm.set_pending(current_id, state.last_action_label)
        if policy_done:
            learn_skill(state.task, state.history, ui)
            state.done = True
        return

    # Skill Library
    skill_actions = get_skill_actions(
        effective_task, ui,
        history=state.history,
        max_steps=max_skill_steps
    ) if max_skill_steps else []

    if skill_actions:
        print(f"[skill] matched — executing {len(skill_actions)} actions")
        for action in skill_actions:
            action = _resolve_action(action, ui, state.memory,
                                     task=state.task, device=device)
            print(f"[skill] action: {action}")
            execute(action, ui, adb_device=device)
            label = _action_label(action, ui)
            state.record(action, ui)
            state.last_action_label = label
            state.last_screen_id    = current_id
        _tm.set_pending(current_id, state.last_action_label)
        return

    # LLM
    prompt = build_prompt(
        effective_task,
        ui,
        history=state.history,
        memory=state.memory,
        sub_goal=sub_goal_desc,
    )
    raw    = generate(prompt, model_path=model_path)
    action = parse_action(raw, ui)

    # ── OPT 13: wait action — AI still generating, pause and re-screenshot ───
    if action.get("action") == "wait":
        warn = action.get("_warn", "")
        print(f"[step] WAIT — {warn or 'AI generating'} — invalidating cache, sleeping {WAIT_ACTION_SLEEP}s")
        invalidate_cache(device)   # Session 16 fix: force fresh dump next step
        time.sleep(WAIT_ACTION_SLEEP)
        state.step += 1
        # Do NOT update last_action_label so transition memory stays clean
        return
    # ─────────────────────────────────────────────────────────────────────────

    action = _resolve_action(action, ui, state.memory,
                             task=state.task, device=device)
    print(f"[llm] action: {action}")
    execute(action, ui, adb_device=device)

    # Invalidate cache after every LLM action — any action can change screen
    # contents while activity stays the same (send button after type, input
    # focus after tap, etc). Matches v0: unconditional invalidate every step.
    invalidate_cache(device)

    label   = _action_label(action, ui)
    outcome = _screen_name(get_top_activity(device))
    state.record(action, ui, outcome_screen=outcome)
    state.last_action_label = label
    state.last_screen_id    = current_id
    _tm.set_pending(current_id, label)

    if action.get("action") == "done":
        learn_skill(state.task, state.history, ui)
        state.done = True


def run(task, device, model_path, max_steps=20, max_skill_steps=0):
    print(f"\n{'#'*56}\n  JARVIS  |  {task}\n  Device : {device}\n{'#'*56}\n")

    hc = health_check()
    print(f"[init] backends: {hc}")
    if not any([hc["phone_b"], hc["proxy"], hc["llamafile"], hc["binary"]]):
        print("[init] WARNING: no LLM backend reachable!")

    _sm.print_stats()
    _tm.print_stats()

    # OPT 12: Create planner
    planner = TaskPlanner(task)

    state                   = AgentState(task=task, max_steps=max_steps)
    state.task_apps         = extract_apps_in_order(task)
    state.app_index         = 0
    state.empty_ui_count    = 0
    state.last_screen_id    = None
    state.last_action_label = None
    print(f"[init] task_apps: {state.task_apps}")
    if max_skill_steps:
        print(f"[init] max_skill_steps: {max_skill_steps} "
              f"(threshold: {int(max_skill_steps*0.8)})")

    while not state.is_finished() and not planner.done():
        try:
            run_step(state, device=device, model_path=model_path,
                     max_skill_steps=max_skill_steps, planner=planner)
        except KeyboardInterrupt:
            print("\n[controller] interrupted")
            break
        except Exception as e:
            print(f"[controller] step error: {e}")
            import traceback; traceback.print_exc()
            # Invalidate cache on error — the action may have partially
            # executed (tap fired) but invalidate_cache never ran.
            # Without this, the next step serves stale pre-action cache.
            try:
                invalidate_cache(device)
            except Exception:
                pass
            time.sleep(3)
            state.step += 1
            continue

        if state.done:
            print(f"\n[controller] done in {state.step} steps")
            break

        time.sleep(LOOP_SLEEP)

//===== TERMUXFILE: test/jarvis-agent/agent_server.py =====
import json
import os
import signal
import subprocess
import threading
import time
from collections import deque
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import urlparse, parse_qs

# ── Pause flag path (must match controller.py) ────────────────────────────────

PAUSE_FLAG = os.path.expanduser("~/workflows/.jarvis_pause")

# ── State ─────────────────────────────────────────────────────────────────────

_state = {
    "running":   False,
    "task":      "",
    "step":      0,
    "done":      False,
    "error":     None,
    "last_log":  "",
    "paused":    False,
}
_state_lock   = threading.Lock()
_log_history  = deque(maxlen=500)
_log_lock     = threading.Lock()
_subscribers  = []
_sub_lock     = threading.Lock()
_proc         = None
_proc_lock    = threading.Lock()

# ── Noise filter ──────────────────────────────────────────────────────────────

_NOISE_PREFIXES = ("llama_", "ggml_", "build:", "sampling params:")

def _is_noise(line: str) -> bool:
    s = line.strip()
    return any(s.startswith(p) for p in _NOISE_PREFIXES)

# ── Log helpers ───────────────────────────────────────────────────────────────

def _push_log(line: str):
    if _is_noise(line):
        return
    with _log_lock:
        _log_history.append(line)
    with _state_lock:
        _state["last_log"] = line
    with _sub_lock:
        dead = []
        for q in _subscribers:
            try:
                q.put_nowait(line)
            except Exception:
                dead.append(q)
        for q in dead:
            _subscribers.remove(q)

# ── Pause / Resume helpers ────────────────────────────────────────────────────

def _is_paused() -> bool:
    return os.path.exists(PAUSE_FLAG)

def _do_pause() -> dict:
    try:
        os.makedirs(os.path.dirname(PAUSE_FLAG), exist_ok=True)
        open(PAUSE_FLAG, "w").close()
        with _state_lock:
            _state["paused"] = True
        _push_log("[agent_server] ⏸ Agent paused")
        return {"paused": True, "flag": PAUSE_FLAG}
    except Exception as e:
        return {"paused": False, "error": str(e)}

def _do_resume() -> dict:
    try:
        if os.path.exists(PAUSE_FLAG):
            os.remove(PAUSE_FLAG)
        with _state_lock:
            _state["paused"] = False
        _push_log("[agent_server] ▶ Agent resumed")
        return {"paused": False, "flag": PAUSE_FLAG}
    except Exception as e:
        return {"paused": False, "error": str(e)}

# ── Controller subprocess ─────────────────────────────────────────────────────

def _run_controller(task: str, device: str, max_steps: int):
    global _proc
    cmd = [
        "python3",
        os.path.expanduser("~/workflows/test/jarvis-agent/agent/controller.py"),
        "--device", device,
        "--task", task,
        "--max-steps", str(max_steps),
    ]
    with _state_lock:
        _state.update({"running": True, "task": task, "step": 0,
                        "done": False, "error": None, "paused": False})

    try:
        with _proc_lock:
            _proc = subprocess.Popen(
                cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                text=True,
                bufsize=1,
            )

        for line in _proc.stdout:
            line = line.rstrip()
            _push_log(line)
            if "STEP" in line and "|" in line:
                try:
                    step = int(line.split("STEP")[1].split("|")[0].strip())
                    with _state_lock:
                        _state["step"] = step
                except Exception:
                    pass
            # Sync paused state from controller log lines
            if "⏸ PAUSED" in line:
                with _state_lock:
                    _state["paused"] = True
            elif line.startswith("[controller]") and "PAUSED" not in line:
                # Only clear paused flag if the controller is actively running
                if _is_paused() is False:
                    with _state_lock:
                        _state["paused"] = False

        _proc.wait()
        rc = _proc.returncode

        with _state_lock:
            _state["running"] = False
            _state["done"]    = True
            _state["paused"]  = False
            if rc != 0 and rc is not None:
                _state["error"] = f"exit code {rc}"

        _push_log(f"[agent_server] Task complete (exit {rc})")

    except Exception as e:
        with _state_lock:
            _state.update({"running": False, "done": True, "error": str(e), "paused": False})
        _push_log(f"[agent_server] ✗ Exception: {e}")

# ── HTTP handler ──────────────────────────────────────────────────────────────

class Handler(BaseHTTPRequestHandler):

    def log_message(self, fmt, *args):
        pass

    def _send_json(self, code: int, obj: dict):
        body = json.dumps(obj).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        try:
            self.wfile.write(body)
        except BrokenPipeError:
            pass

    def _read_json(self) -> dict:
        length = int(self.headers.get("Content-Length", 0))
        return json.loads(self.rfile.read(length)) if length else {}

    def do_POST(self):
        path = urlparse(self.path).path

        if path == "/run":
            body = self._read_json()
            with _state_lock:
                already = _state["running"]
            if already:
                self._send_json(409, {"error": "already running"})
                return
            task      = body.get("task", "")
            device    = body.get("device", "192.168.29.48:40657")
            max_steps = int(body.get("max_steps", 20))
            # Clear any stale pause flag on new run
            _do_resume()
            threading.Thread(
                target=_run_controller,
                args=(task, device, max_steps),
                daemon=True
            ).start()
            self._send_json(200, {"ok": True, "task": task})

        elif path == "/stop":
            with _proc_lock:
                p = _proc
            if p and p.poll() is None:
                try:
                    os.killpg(os.getpgid(p.pid), signal.SIGTERM)
                except Exception:
                    p.terminate()
                _push_log("[agent_server] Stop signal sent")
            with _state_lock:
                _state.update({"running": False, "done": True, "paused": False})
            # Also remove pause flag so it doesn't linger
            try:
                if os.path.exists(PAUSE_FLAG):
                    os.remove(PAUSE_FLAG)
            except Exception:
                pass
            self._send_json(200, {"ok": True})

        elif path == "/pause":
            result = _do_pause()
            self._send_json(200, {"ok": True, **result})

        elif path == "/resume":
            result = _do_resume()
            self._send_json(200, {"ok": True, **result})

        else:
            self._send_json(404, {"error": "not found"})

    def do_GET(self):
        parsed = urlparse(self.path)
        path   = parsed.path
        params = parse_qs(parsed.query)

        if path == "/status":
            with _state_lock:
                s = dict(_state)
            # Always reflect the real file state for paused
            s["paused"] = _is_paused()
            self._send_json(200, s)

        elif path == "/health":
            self._send_json(200, {"ok": True, "paused": _is_paused()})

        elif path == "/clipboard":
            # Reads system clipboard via termux-clipboard-get.
            # controller.py calls this instead of subprocess because ADB shell
            # user cannot access Termux clipboard — only Termux user can.
            # Retry loop — Copy tap may not have propagated to clipboard yet.
            text = ""
            last_err = None
            for attempt in range(10):
                try:
                    result = subprocess.run(
                        ["termux-clipboard-get"],
                        capture_output=True, text=True, timeout=5
                    )
                    text = result.stdout.strip()
                    if text:
                        break
                except Exception as e:
                    last_err = str(e)
                time.sleep(0.5)
            if text:
                self._send_json(200, {"ok": True, "text": text, "attempts": attempt + 1})
            else:
                self._send_json(200, {"ok": False, "text": "",
                                      "error": last_err or "clipboard empty after 10 attempts"})

        elif path == "/logs/tail":
            n = min(int(params.get("n", ["50"])[0]), 100)
            with _log_lock:
                lines = list(_log_history)[-n:]
            body = json.dumps({"lines": lines}).encode()
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            try:
                self.wfile.write(body)
            except BrokenPipeError:
                pass

        elif path == "/logs":
            import queue as qmod
            q = qmod.Queue()
            with _sub_lock:
                _subscribers.append(q)

            self.send_response(200)
            self.send_header("Content-Type", "text/event-stream")
            self.send_header("Cache-Control", "no-cache")
            self.send_header("Connection", "keep-alive")
            self.end_headers()

            try:
                while True:
                    try:
                        line = q.get(timeout=15)
                        msg  = f"data: {line}\n\n".encode()
                        self.wfile.write(msg)
                        self.wfile.flush()
                    except Exception:
                        try:
                            self.wfile.write(b": ping\n\n")
                            self.wfile.flush()
                        except Exception:
                            break
            except Exception:
                pass
            finally:
                with _sub_lock:
                    if q in _subscribers:
                        _subscribers.remove(q)

        else:
            self._send_json(404, {"error": "not found"})

if __name__ == "__main__":
