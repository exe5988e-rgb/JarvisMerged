 157        override suspend fun generateReply(prompt: String): String {

 160                return "AI engine is initializing... Please wait a moment."

 163            if (chatService == null && codeService == null) {

 167            }
 168            
 169            val isCodeRequest = isCodeRelated(prompt)

 172                try {
 173                    // ✅ FIXED: Using when expression which always has all branches covered

 175                        when {

 178                                val formattedPrompt = "### Instruction:\n$prompt\n\n### Response:\n"
 179                                codeService!!.generate(formattedPrompt, maxTokens = 256, temperature = 0.2f)
 180                            }
 181                            chatService != null -> {
 182                                Log.d(TAG, "Using chat model for generation")

 184                            }

 189                    }

 195                    "Request cancelled."
 196                } catch (e: Exception) {
