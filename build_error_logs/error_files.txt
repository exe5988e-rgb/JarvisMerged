  34                log("Iteration $iteration - State: $state")
  35                state = when (state) {
  36                    AutoBuildState.WAITING_FOR_RESPONSE -> { delay(2000); AutoBuildState.DOWNLOAD_AI_OUTPUT }
  37                    AutoBuildState.DOWNLOAD_AI_OUTPUT -> {
  38                        fileManager.deleteOldAiOutput()
  39                        val success = uiWatcher.handleDownloadAiOutput()
  40                        if (success) AutoBuildState.COPY_TO_AUTOMATION_DIR
  41                        else { delay(2000); AutoBuildState.DOWNLOAD_AI_OUTPUT }
  42                    }
  43                    AutoBuildState.COPY_TO_AUTOMATION_DIR -> {
  44                        val success = fileManager.copyAiOutputToAutomationDir()
  45                        if (success) AutoBuildState.TRIGGER_BUILD
  46                        else { delay(2000); AutoBuildState.COPY_TO_AUTOMATION_DIR }
  47                    }
  48                    AutoBuildState.TRIGGER_BUILD -> AutoBuildState.WAITING_FOR_BUILD
  49                    AutoBuildState.WAITING_FOR_BUILD -> {
  50 ❌                     when (termuxBridge.triggerBuildAndWait(fileManager)) {
  51 ❌                         BuildResult.SUCCESS -> AutoBuildState.BUILD_SUCCEEDED
  52 ❌                         BuildResult.FAILURE -> AutoBuildState.ATTACHING_ERROR_REPORT
  53 ❌                         BuildResult.TIMEOUT -> { notifier.error("Build timeout"); return }
  54 ❌                         BuildResult.ERROR   -> { notifier.error("Build error");   return }
  55                        }
  56                    }
  57                    AutoBuildState.BUILD_SUCCEEDED -> { log("Pipeline completed after $iteration iterations"); notifier.success(iteration); return }
  58                    AutoBuildState.ATTACHING_ERROR_REPORT -> {
  59                        val success = uiWatcher.handleBuildFailure()
  60                        if (success) { iteration++; AutoBuildState.WAITING_FOR_RESPONSE }
  61                        else { delay(2000); AutoBuildState.ATTACHING_ERROR_REPORT }
  62                    }
  63                    else -> { log("Unknown state: $state", isError = true); notifier.error("Unknown state"); return }
  64                }
  65                updateState(iteration, state)
  66                delay(500)
  67            }
  68            if (stopRequested) { log("Loop stopped by user"); notifier.error("Stopped by user") }

  72        // ── Session 20: Agent loop ─────────────────────────────────────────────
  73        // Triggered when AutoBuildService detects agent_task.txt on Phone B sdcard.
  74        // Flow: AGENT_SENDING_DUMP → AGENT_WAITING_FOR_RESPONSE →
  75        //       AGENT_DOWNLOAD_OUTPUT → AGENT_STAGING_OUTPUT → AGENT_LOOP_DONE
  76        // Single-shot — no iteration loop. Phone A polls pull_from_bridge after.
  77    
  78        suspend fun runAgentLoop(task: String, dumpName: String) {
  79            log("[agent] Starting agent loop: task=${task.take(60)} dump=$dumpName")
  80            var state = AutoBuildState.AGENT_SENDING_DUMP
  81            updateState(0, state)
  82    
  83            while (!stopRequested) {
  84                log("[agent] State: $state")
  85                state = when (state) {
  86    
  87                    AutoBuildState.AGENT_SENDING_DUMP -> {
  88                        if (!fileManager.agentDumpExists(dumpName)) {
  89                            log("[agent] Dump file not found: $dumpName — waiting...", isError = true)
  90                            delay(2000)
  91                            AutoBuildState.AGENT_SENDING_DUMP
  92                        } else {
  93                            val success = uiWatcher.handleSendDump(task, dumpName)
  94                            if (success) {
  95                                log("[agent] Dump sent to Claude, waiting for response...")
  96                                AutoBuildState.AGENT_WAITING_FOR_RESPONSE
  97                            } else {
  98                                log("[agent] Failed to send dump, retrying...", isError = true)
  99                                delay(3000)
 100                                AutoBuildState.AGENT_SENDING_DUMP
 101                            }
 102                        }
 103                    }
 104    
 105                    AutoBuildState.AGENT_WAITING_FOR_RESPONSE -> {
 106                        // Give Claude time to process and generate the output file
 107                        log("[agent] Waiting for Claude to generate output...")
 108                        delay(5000)
 109                        AutoBuildState.AGENT_DOWNLOAD_OUTPUT
 110                    }
 111    
 112                    AutoBuildState.AGENT_DOWNLOAD_OUTPUT -> {
 113                        fileManager.deleteOldAgentOutput()
 114                        log("[agent] Downloading ai-output.txt...")
 115                        val success = uiWatcher.handleDownloadAiOutput()
 116                        if (success) {
 117                            AutoBuildState.AGENT_STAGING_OUTPUT
 118                        } else {
 119                            log("[agent] Download failed, retrying...", isError = true)
 120                            delay(3000)
 121                            AutoBuildState.AGENT_DOWNLOAD_OUTPUT
 122                        }
 123                    }
 124    
 125                    AutoBuildState.AGENT_STAGING_OUTPUT -> {
 126                        log("[agent] Staging output for Phone A...")
 127                        val success = fileManager.stageOutputForPhoneA()
 128                        if (success) {
 129                            AutoBuildState.AGENT_LOOP_DONE
 130                        } else {
 131                            log("[agent] Staging failed, retrying...", isError = true)
 132                            delay(2000)
 133                            AutoBuildState.AGENT_STAGING_OUTPUT
 134                        }
 135                    }
 136    
 137                    AutoBuildState.AGENT_LOOP_DONE -> {
 138                        log("[agent] ✅ Done — ai-output.txt staged for Phone A")
 139                        notifier.agentSuccess()
 140                        return
 141                    }
 142    
 143                    else -> {
 144                        log("[agent] Unexpected state: $state", isError = true)
 145                        notifier.error("[agent] Unexpected state")
 146                        return
 147                    }
 148                }
 149                updateState(0, state)
 150                delay(500)
 151            }
 152            log("[agent] Stopped by user request")
 153            notifier.error("[agent] Stopped by user")
 154        }

 156        private fun updateState(iteration: Int, state: AutoBuildState) {
 157            AutoBuildService.currentState     = state
 158            AutoBuildService.currentIteration = iteration
 159            AutoBuildService.onStatusUpdate?.invoke(iteration, state)
 160            notifier.update(iteration, state)
 161        }

 164            if (isError) Log.e(TAG, msg) else Log.d(TAG, msg)
 165            AutoBuildService.onLogLine?.invoke(msg, isError)
 166        }
