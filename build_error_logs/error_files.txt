 865    fun getLocalIpAddress(): String {
 866        try {
 867            val interfaces = NetworkInterface.getNetworkInterfaces()
 868            while (interfaces.hasMoreElements()) {
 869                val networkInterface = interfaces.nextElement()
 870                val addresses = networkInterface.inetAddresses
 871                while (addresses.hasMoreElements()) {
 872                    val address = addresses.nextElement()
 873                    if (!address.isLoopbackAddress && address is java.net.Inet4Address) {
 874                        return address.hostAddress ?: "Unknown"
 875                    }
 876                }
 877            }
 878        } catch (e: Exception) {
 879            e.printStackTrace()
 880        }
 881        return "Unknown"
 882    }
