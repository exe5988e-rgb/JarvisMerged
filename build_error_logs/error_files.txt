121	private fun parseTimeToMs(timeStr: String): Long {
122	    return try {
123	        val cal = Calendar.getInstance()
124	        val parts = timeStr.split(":")
125	        cal.set(Calendar.HOUR_OF_DAY, parts[0].toInt())
126	        cal.set(Calendar.MINUTE, parts[1].toInt())
127	        cal.set(Calendar.SECOND, 0)
