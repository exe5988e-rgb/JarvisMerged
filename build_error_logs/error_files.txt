 864    fun getLocalIpAddress(): String {
 865        try {
 866            val interfaces = NetworkInterface.getNetworkInterfaces()
 867            while (interfaces.hasMoreElements()) {
 868                val networkInterface = interfaces.nextElement()
 869                val addresses = networkInterface.inetAddresses
 870                while (addresses.hasMoreElements()) {
 871                    val address = addresses.nextElement()
 872                    if (!address.isLoopbackAddress && address is java.net.Inet4Address) {
 873                        return address.hostAddress ?: "Unknown"
 874                    }
 875                }
 876            }
 877        } catch (e: Exception) {
 878            e.printStackTrace()
 879        }
 880        return "Unknown"
 881    }
