 130 ❌ def run_agent_task(task: str) -> str:
 131 ❌     try:
 132 ❌         from agent.state               import AgentState
 133 ❌         from perception.screen_capture import capture_screen
 134 ❌         from perception.ui_mock_parser  import parse_ui
 135 ❌         from reasoning.prompt_builder   import build_prompt
 136 ❌         from reasoning.llm_client       import generate
 137 ❌         from reasoning.action_parser    import parse_action
 138 ❌         from executor.adb_executor      import execute
 139    
 140 ❌         STATUS_FILE.write_text("running")
 141 ❌         state = AgentState(task=task, max_steps=20)
 142 ❌         last_action_label = "none"
 143    
 144 ❌         while not state.is_finished():
 145 ❌             try:
 146 ❌                 img    = capture_screen(adb_device=ADB_DEVICE)
 147 ❌                 ui     = parse_ui(img)
 148 ❌                 prompt = build_prompt(task, ui, history=state.history)
 149 ❌                 out    = generate(prompt)
 150 ❌                 action = parse_action(out)
 151 ❌                 execute(action, ui, adb_device=ADB_DEVICE)
 152 ❌                 state.record(action, ui)
 153    
 154 ❌                 if action.get("action") == "done":
 155 ❌                     state.done = True
 156 ❌                     break
 157    
 158 ❌                 act = action.get("action", "?")
 159 ❌                 if act == "tap" and action.get("target"):
 160 ❌                     tid   = action.get("target")
 161 ❌                     label = next((e["text"] for e in ui if e["id"] == tid), f"item {tid}")
 162 ❌                     last_action_label = f"tapped {label}"
 163 ❌                 elif act == "type":
 164 ❌                     last_action_label = f"typed '{action.get('text','')}'"
 165 ❌                 else:
 166 ❌                     last_action_label = act
 167    
 168 ❌                 time.sleep(1.5)
 169    
 170 ❌             except Exception as e:
 171 ❌                 print(f"[bridge] step error: {e}")
 172 ❌                 time.sleep(3)
 173 ❌                 state.step += 1
 174 ❌                 continue
 175    
 176 ❌         if state.done:
 177 ❌             result = f"Done. {task} completed in {state.step} steps."
 178 ❌         elif state.step >= state.max_steps:
 179 ❌             result = f"Reached maximum steps. Last action: {last_action_label}."
 180 ❌         else:
 181 ❌             result = f"Task stopped. Last action: {last_action_label}."
 182    
 183 ❌         STATUS_FILE.write_text("ready")
 184 ❌         return result

 186 ❌     except Exception as e:
 187 ❌         STATUS_FILE.write_text("error")
