 172        /**
 173         * Dual-mode download detection:
 174         *
 175         * Mode A (primary) — name diff: any non-empty file not in beforeSnapshot.
 176         * Mode B (fallback) — timestamp: newest non-empty file modified after clickTimestamp.
 177         *
 178         * Mode B catches the ultra-fast download case where the file lands before
 179         * the snapshot was taken, making it invisible to name-diff.

 181        private suspend fun waitForNewDownload(
 182            dir: File,
 183            beforeSnapshot: Set<String>,
 184            clickTimestamp: Long
 185        ): File? = withContext(Dispatchers.IO) {
 186            val startTime = System.currentTimeMillis()
 187            while (System.currentTimeMillis() - startTime < MAX_DOWNLOAD_WAIT_MS) {
 188                val allFiles = dir.listFiles()?.filter { it.length() > 0 } ?: emptyList()
 189    
 190                // Mode A: name diff
 191                val newByName = allFiles.firstOrNull { it.name !in beforeSnapshot }
 192                if (newByName != null) {
 193                    Log.d(TAG, "Download detected (name-diff): ${newByName.name}")
 194                    return@withContext newByName
 195                }
 196    
 197                // Mode B: timestamp — newest file modified after click
 198                val newByTimestamp = allFiles
 199                    .filter { it.lastModified() > clickTimestamp }
 200                    .maxByOrNull { it.lastModified() }
 201                if (newByTimestamp != null) {
 202                    Log.d(TAG, "Download detected (timestamp): ${newByTimestamp.name}")
 203                    return@withContext newByTimestamp
 204                }
 205    
 206                delay(POLL_INTERVAL_MS)
 207            }
 208            null
 209        }

 301        private fun subtreeContainsKeyword(
 302            node: AccessibilityNodeInfo,
 303            keywords: List<String>
 304        ): Boolean {
 305            val text = node.text?.toString() ?: ""

 313                child.recycle()

 499        private fun findClickableAncestorByFuzzyText(
 500            root: AccessibilityNodeInfo,
 501            keyword: String
 502        ): AccessibilityNodeInfo? {
 503            val nodes = root.findAccessibilityNodeInfosByText(keyword)
 504            for (node in nodes) {
 505                if (node.text?.toString()?.contains(keyword, ignoreCase = true) == true) {
 506                    if (node.isClickable) return node
 507                    var parent = node.parent
 508                    while (parent != null) {
 509                        if (parent.isClickable) return parent
 510                        val next = parent.parent
 511                        if (next == parent) break
 512                        parent = next
 513                    }
 514                }
 515            }
 516            return null
 517        }

 521            keyword: String
 522        ): AccessibilityNodeInfo? {
 523            val desc = root.contentDescription?.toString() ?: ""

 527                val result = findNodeByContentDescriptionFuzzy(child, keyword)
 528                if (result != null) return result
