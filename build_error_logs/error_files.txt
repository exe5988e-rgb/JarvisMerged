 683    private fun getLocalIpAddress(): String {
 684        return try {
 685            NetworkInterface.getNetworkInterfaces()?.toList()
 686                ?.flatMap { it.inetAddresses.toList() }
 687                ?.firstOrNull { !it.isLoopbackAddress && it.hostAddress?.contains(':') == false }
 688                ?.hostAddress ?: "Unknown"
 689        } catch (e: Exception) {
 690            "Unknown"
 691        }
 692    }
