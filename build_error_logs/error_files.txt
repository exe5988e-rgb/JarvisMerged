 323    private fun formatFTime(seconds: Long): String {
 324        val h = seconds / 3600
 325        val m = (seconds % 3600) / 60
 326        val s = seconds % 60
 327        return if (h > 0) "%d:%02d:%02d".format(h, m, s)
 328        else "%02d:%02d".format(m, s)
